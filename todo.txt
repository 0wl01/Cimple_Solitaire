void test_split_deck() {
    Deck *d1 = create_deck(5);
    Deck *d2 = create_deck(5);
    
    // Mete cartas de valor 1, 2, 3, 4, 5
    for(uint8_t i = 1; i <= 5; i++) {
        push(d1, (Card){.values = {.value = i}});
    }
    
    // Tenta dividir a partir do índice 2 (deve mover os valores 3, 4 e 5)
    CU_ASSERT_TRUE(split_deck(d1, d2, 2));
    
    CU_ASSERT_EQUAL(d1->top, 2); // Ficam 2 cartas na origem
    CU_ASSERT_EQUAL(d2->top, 3); // Foram 3 para o destino
    CU_ASSERT_EQUAL(d2->cards[0].values.value, 3); // A primeira do destino é o 3
    
    eliminate_deck(&d1);
    eliminate_deck(&d2);
}
